    330: /**
    331: * Check that the certificate is trusted: OpenSSL builds and verifies its
    332: * chain up to a certificate of the given CA files.
    333: *
    334: * Every certificate of the given files is trusted, and the chain is built
    335: * from these certificates only. OpenSSL verifies the signature of each
    336: * link, the validity dates of every certificate of the chain at the
    337: * current date, that each issuer is a CA (basic constraints, key usage,
    338: * path length) and that the chain ends at a self-signed certificate of
    339: * the given files. A chain that stops at an intermediate CA is rejected:
    340: * to trust the certificates issued by an intermediate CA, give its
    341: * certificate and the certificates up to its root (in one file or in
    342: * several). A certificate issued by another intermediate CA of the same
    343: * root is then rejected, since the chain cannot be built without that
    344: * other intermediate certificate.
    345: *
    346: * @param string|array $caCertPaths The path, or the list of paths, of PEM
    347: * files that hold the trusted CA certificates.
    348: *
    349: * @return boolean TRUE if the chain of the certificate has been verified,
    350: * FALSE otherwise.
    351: *
    352: * @throws Exception
    353: */
    354: public function isTrustedBy($caCertPaths)
    355: {
    356: $caCertPaths = (array) $caCertPaths;
    357: if (empty($caCertPaths))
    358: {
    359: throw new Exception("No CA certificate file to verify the certificate chain!");
    360: }
    361: foreach ($caCertPaths as $caCertPath)
    362: {
    363: if (!is_string($caCertPath) || !is_file($caCertPath) || !is_readable($caCertPath))
    364: {
    365: throw new Exception("Cannot read CA certificate file! File: " . (is_string($caCertPath) ? $caCertPath : gettype($caCertPath)));
    366: }
    367: }
    368:
    369: // Only the given CA certificates may be trusted, not those of the
    370: // system. When the list given to openssl_x509_checkpurpose() holds no
    371: // directory, PHP adds the default CA directory of OpenSSL (that of
    372: // the system, or the one named by the environment variable
    373: // SSL_CERT_DIR) to the trusted certificates, and the default CA file
    374: // when the list holds no file: any public CA would then be a trust
    375: // anchor. An empty directory, created for this call, is therefore
    376: // always added to the list. This guard must not be removed.
    377: $emptyDir = self::createEmptyDirectory();
    378: try
    379: {
    380: // X509_PURPOSE_ANY: the chain is verified, but no purpose is
    381: // required from the certificate (see canSign() for its key
    382: // usage). openssl_x509_checkpurpose() returns -1 on error: only
    383: // TRUE is a success.
    384: $caInfo = array_merge(array_values($caCertPaths), [$emptyDir]);
    385: return true === openssl_x509_checkpurpose($this->cert, X509_PURPOSE_ANY, $caInfo);
    386: }
    387: finally
    388: {
    389: @rmdir($emptyDir);
    390: }
    391: }

    416: /**
    417: * Check the origin of the certificate.
    418: *
    419: * Verify that the X.509 certificate has been issued by the given CA
    420: * certificate: OpenSSL verifies the signature of the certificate with the
    421: * public key of the CA certificate, the issuer name of the certificate
    422: * must be the subject name of the CA certificate, and the CA certificate
    423: * must be a CA ("CA:TRUE" basic constraint and, if it has a "keyUsage"
    424: * extension, "keyCertSign").
    425: *
    426: * Only this link is checked: neither the validity dates nor the chain of
    427: * the CA certificate. Use `isTrustedBy()` to verify the whole chain.
    428: *
    429: * NOTE: More than one CA certificate can give a positive result,
    430: * some CAs re-issue signing certificates after having only
    431: * changed the expiration dates.
    432: * Note that it also works with self-signed CA certificates. In this
    433: * case, pass the X.509 certificate itself to the function.
    434: *
    435: * @param string $caCert The certificate (in PEM format) corresponding to
    436: * the private key that is supposed to be used by the CA to sign the X.509
    437: * certificate.
    438: *
    439: * @return boolean TRUE if origin of the X.509 certificate has been
    440: * validated, FALSE otherwise.
    441: *
    442: * @throws Exception
    443: */
    444: public function isValidCA($caCert)
    445: {
    446: if (!is_string($caCert) || false === @openssl_x509_read($caCert))
    447: {
    448: throw new Exception("Invalid CA certificate!");
    449: }
    450:
    451: $caInfo = openssl_x509_parse($caCert);
    452: if ($caInfo === false)
    453: {
    454: throw new Exception("Failed to extract X.509 information from CA certificate!");
    455: }
    456:
    457: // The CA certificate must be allowed to issue certificates.
    458: $basicConstraints = isset($caInfo['extensions']['basicConstraints']) ? $caInfo['extensions']['basicConstraints'] : '';
    459: if (!in_array('CA:TRUE', array_map('trim', explode(',', $basicConstraints)), true))
    460: {
    461: return false;
    462: }
    463: if (isset($caInfo['extensions']['keyUsage']))
    464: {
    465: $keyUsage = array_map('trim', explode(',', $caInfo['extensions']['keyUsage']));
    466: if (!in_array('Certificate Sign', $keyUsage, true))
    467: {
    468: return false;
    469: }
    470: }
    471:
    472: // The certificate must name the CA as its issuer.
    473: if ($this->x509Info['issuer'] !== $caInfo['subject'])
    474: {
    475: return false;
    476: }
    477:
    478: // The signature of the certificate must verify with the public key of
    479: // the CA. openssl_x509_verify() returns 1 (valid), 0 (invalid) or -1
    480: // (error): only 1 is a success.
    481: return 1 === openssl_x509_verify($this->cert, $caCert);
    482: }

    485: /**
    486: * Convert PEM encoded X.509 certificate to DER encoding.
    487: *
    488: * @param string $pemCertificate PEM encoded X.509 certificate.
    489: *
    490: * @return string|false DER encoded X.509 certificate on success,
    491: * FALSE on failure.

    493: static public function pemToDer($pemCertificate)
    494: {
    495: if (!is_string($pemCertificate))
    496: {
    497: // Invalid parameter.
    498: return false;
    499: }
    500:
    501: $pattern = '/(-----((BEGIN)|(END)) CERTIFICATE-----)/';
    502: $base64Content = preg_split($pattern, $pemCertificate);
    503: if (!isset($base64Content[1]))
    504: {
    505: // Invalid PEM certificate.
    506: return false;
    507: }
    508:
    509: // Returns the DER encoded certificate.
    510: return base64_decode($base64Content[1]);
    511: }

    514: /**
    515: * Convert DER encoded X.509 certificate to PEM encoding.
    516: *
    517: * The base64 is written in lines of 64 characters (RFC 7468): the PEM is
    518: * the one OpenSSL writes for the same certificate.
    519: *
    520: * @param string $derCertificate DER encoded X.509 certificate.
    521: *
    522: * @return string|false PEM encoded X.509 certificate on success,
    523: * FALSE on failure.
    524: */
    525: static public function derToPem($derCertificate)
    526: {
    527: if (!is_string($derCertificate))
    528: {
    529: // Invalid parameter.
    530: return false;
    531: }
    532:
    533: $b64DerCert = base64_encode($derCertificate);
    534: $pem = "-----BEGIN CERTIFICATE-----\n";
    535: // Lines of 64 characters, as RFC 7468 (section 2) requires and
    536: // OpenSSL writes.
    537: $pem .= implode("\n", str_split($b64DerCert, 64)) . "\n";
    538: $pem .= "-----END CERTIFICATE-----\n";
    539:
    540: return $pem;
    541: }

    662:
    663: /**
    664: * The chain of the certificate, from it to its root, built and verified
    665: * from the given CA certificates as OpenSSL does for `isTrustedBy()`, at
    666: * the current date or at another one; NULL if the certificate is not
    667: * trusted.
    668: *
    669: * Every given certificate is trusted, and the chain is built from these
    670: * certificates only (and the `$untrusted` ones), never from those of the
    671: * system. A certificate is taken as the issuer of another one when its
    672: * subject is the issuer of the other one (names compared in the canonical
    673: * form of OpenSSL), when the authority key identifier of the other one
    674: * matches it, and when its key suits the signature algorithm of the other
    675: * one. Then:
    676: *
    677: * - OpenSSL verifies the signature of each link (but the root's own);
    678: * - every certificate of the chain, the root included, is valid at the
    679: * date of the verification;
    680: * - every issuer is a CA: basic constraints with "CA:TRUE" and, if it has
    681: * a key usage, "keyCertSign"; the root may also be a CA by its key
    682: * usage alone, or by being self-signed and of version 1;
    683: * - the path length of each CA is respected;
    684: * - no certificate has a critical extension the verification does not
    685: * handle, nor a repeated or malformed extension.
    686: *
    687: * A chain that stops at an intermediate CA is rejected: the root must be
    688: * one of the CA certificates. Name constraints are not enforced: a CA
    689: * certificate that has them is rejected, where OpenSSL would enforce
    690: * them.
    691: *
    692: * @param string|array $caCerts The CA certificates: paths of PEM files,
    693: * or PEM contents, each of which may hold several certificates.
    694: *
    695: * @param array $untrusted Certificates (X509Cert) the chain may be built
    696: * with, after the CA certificates, without being trusted: the
    697: * intermediate CAs a signature embeds.
    698: *
    699: * @param int|null $at The UNIX time the certificates must be valid at;
    700: * NULL for the current time.
    701: *
    702: * @return array|null The chain (X509Cert), or NULL.
    703: *
    704: * @throws Exception If a CA certificate file cannot be read.
    705: */
    706: public function getTrustedChain($caCerts, array $untrusted = [], $at = null)
    707: {
    708: $sources = (array) $caCerts;
    709: if (empty($sources))
    710: {
    711: throw new Exception("No CA certificate file to verify the certificate chain!");
    712: }
    713: $trusted = [];
    714: foreach ($sources as $source)
    715: {
    716: $pem = self::readPemSource($source);
    717: if (null === $pem)
    718: {
    719: throw new Exception("Cannot read CA certificate file! File: " . (is_string($source) ? $source : gettype($source)));
    720: }
    721: // A file OpenSSL cannot load as a whole trusts nothing.
    722: if (!preg_match_all('/-----BEGIN CERTIFICATE-----.*?-----END CERTIFICATE-----/s', $pem, $blocks))
    723: {
    724: return null;
    725: }
    726: foreach ($blocks[0] as $block)
    727: {
    728: try
    729: {
    730: $trusted[] = new X509Cert($block);
    731: }
    732: catch (Exception $e)
    733: {
    734: return null;
    735: }
    736: }
    737: }
    738:
    739: $time = null === $at ? time() : (int) floor($at);
    740: $chain = $this->buildChain($trusted, $untrusted, $time);
    741:
    742: return null !== $chain && self::verifyChain($chain, $time) ? $chain : null;
    743: }

    746: /**
    747: * The content of a PEM source: a path of a file, or PEM content.
    748: *
    749: * @return string|null The content, or NULL if the file cannot be read.
    750: */
    751: private static function readPemSource($source)
    752: {
    753: if (!is_string($source))
    754: {
    755: return null;
    756: }
    757: if (false !== strpos($source, '-----BEGIN'))
    758: {
    759: return $source;
    760: }
    761: if (!is_file($source) || !is_readable($source))
    762: {
    763: return null;
    764: }
    765: $content = file_get_contents($source);
    766:
    767: return false === $content ? null : $content;
    768: }

    771: /**
    772: * Build the chain of the certificate from the trusted certificates, as
    773: * OpenSSL does: from the certificate, take as issuer the first trusted
    774: * certificate, then the first untrusted one, that may have issued it and
    775: * is valid at the given time (or else the last one that may have issued
    776: * it), until a self-signed certificate, which must be a trusted one.
    777: *
    778: * @return array|null The chain, from this certificate to its root, or NULL.
    779: */
    780: private function buildChain(array $trusted, array $untrusted, $time)
    781: {
    782: $known = array_merge($trusted, $untrusted);
    783: $chain = [$this];
    784: for (;;)
    785: {
    786: $last = $chain[count($chain) - 1];
    787: if ($last->isSelfSigned())
    788: {
    789: foreach ($trusted as $cert)
    790: {
    791: if ($cert->getDer() === $last->getDer())
    792: {
    793: return $chain;
    794: }
    795: }
    796: return null;
    797: }
    798: if (count($chain) >= self::MAX_CHAIN_LENGTH)
    799: {
    800: return null;
    801: }
    802: $issuer = null;
    803: $lastCandidate = null;
    804: foreach ($known as $cert)
    805: {
    806: if ($last->mayBeIssuedBy($cert))
    807: {
    808: $lastCandidate = $cert;
    809: if (null === $issuer && $cert->isValidAt($time))
    810: {
    811: $issuer = $cert;
    812: }
    813: }
    814: }
    815: $issuer = null !== $issuer ? $issuer : $lastCandidate;
    816: if (null === $issuer)
    817: {
    818: return null;
    819: }
    820: $chain[] = $issuer;
    821: }
    822: }

    825: /**
    826: * Verify a chain built by buildChain(), as OpenSSL does (see
    827: * getTrustedChain()).
    828: */
    829: private static function verifyChain(array $chain, $time)
    830: {
    831: $pathLength = 0;
    832: foreach ($chain as $i => $cert)
    833: {
    834: if (!$cert->isValidAt($time))
    835: {
    836: return false;
    837: }
    838: // openssl_x509_verify() returns 1 (valid), 0 (invalid) or -1
    839: // (error): only 1 is a success.
    840: if (isset($chain[$i + 1]) && 1 !== openssl_x509_verify($cert->cert, $chain[$i + 1]->cert))
    841: {
    842: return false;
    843: }
    844:
    845: $der = $cert->der();
    846: if ($der['invalidExtension'])
    847: {
    848: return false;
    849: }
    850: foreach ($der['extensions'] as $oid => $extension)
    851: {
    852: if ($extension['critical'] && !in_array($oid, self::HANDLED_CRITICAL_EXTENSIONS, true))
    853: {
    854: return false;
    855: }
    856: }
    857: if (0 === $i)
    858: {
    859: continue;
    860: }
    861:
    862: if (isset($der['extensions'][self::OID_NAME_CONSTRAINTS]))
    863: {
    864: return false;
    865: }
    866: // An intermediate CA must be one by its basic constraints; the
    867: // root may be one otherwise.
    868: $kind = $cert->caKind();
    869: if (0 === $kind || (isset($chain[$i + 1]) && 1 !== $kind))
    870: {
    871: return false;
    872: }
    873: // The path length of a CA counts the CAs below it, but the
    874: // self-issued ones.
    875: $limit = null === $der['basicConstraints'] ? null : $der['basicConstraints']['pathLength'];
    876: if ($i > 1 && null !== $limit && $pathLength > $limit)
    877: {
    878: return false;
    879: }
    880: if (!$cert->isSelfIssued())
    881: {
    882: $pathLength++;
    883: }
    884: }
    885:
    886: return true;
    887: }
    888:
    889:
    890: /**
    891: * Indicates whether the certificate may have been issued by another one,
    892: * as OpenSSL tells it to build a chain (`ossl_x509_likely_issued()`): the
    893: * issuer of the certificate is the subject of the other one, its
    894: * authority key identifier, if any, matches the other one, and the key of
    895: * the other one suits its signature algorithm. Neither the signature nor
    896: * the key usage of the other one are checked here.
    897: */
    898: private function mayBeIssuedBy(X509Cert $issuer)
    899: {
    900: $der = $this->der();
    901: $issuerDer = $issuer->der();
    902: if ($der['invalidExtension'] || $issuerDer['invalidExtension'])
    903: {
    904: return false;
    905: }
    906: if (!self::sameName($der['issuer'], $issuerDer['subject']))
    907: {
    908: return false;
    909: }
    910: $akid = $der['authorityKeyId'];
    911: if (null !== $akid)
    912: {
    913: if (null !== $akid['keyId'] && null !== $issuerDer['subjectKeyId'] && $akid['keyId'] !== $issuerDer['subjectKeyId'])
    914: {
    915: return false;
    916: }
    917: if (null !== $akid['serial'] && $akid['serial'] !== $issuerDer['serial'])
    918: {
    919: return false;
    920: }
    921: if (null !== $akid['issuer'] && !self::sameName($akid['issuer'], $issuerDer['issuer']))
    922: {
    923: return false;
    924: }
    925: }
    926: $keyTypes = isset(self::SIGNATURE_KEY_TYPES[$der['signatureAlgorithm']]) ? self::SIGNATURE_KEY_TYPES[$der['signatureAlgorithm']] : [];
    927:
    928: return in_array($issuer->keyType(), $keyTypes, true);
    929: }

    964: /**

    966: * (`X509_check_ca()`): 0 if it is not, 1 by its basic constraints, 3 by
    967: * being self-signed and of version 1, 4 by its key usage alone.
    968: */
    969: private function caKind()
    970: {
    971: $der = $this->der();
    972: if (null !== $der['keyUsage'] && empty($der['keyUsage'][self::KEY_USAGE_KEY_CERT_SIGN]))
    973: {
    974: return 0;
    975: }
    976: if (null !== $der['basicConstraints'])
    977: {
    978: return $der['basicConstraints']['ca'] ? 1 : 0;
    979: }
    980: if (1 === $der['version'] && $this->isSelfSigned())
    981: {

    983: }
    984:
    985: return null !== $der['keyUsage'] ? 4 : 0;

    989: /**
    990: * The type of the public key: "rsa", "ec", "dsa", or NULL for another
    991: * one.
    992: */
    993: private function keyType()
    994: {
    995: $key = @openssl_pkey_get_public($this->cert);
    996: $details = false === $key ? false : @openssl_pkey_get_details($key);
    997: if (false === $details)
    998: {
    999: return null;
    1000: }
    1001: switch ($details['type'])
    1002: {
    1003: case OPENSSL_KEYTYPE_RSA:
    1004: return 'rsa';
    1005: case OPENSSL_KEYTYPE_EC:
    1006: return 'ec';
    1007: case OPENSSL_KEYTYPE_DSA:
    1008: return 'dsa';
    1009: default:
    1010: return null;
    1011: }
    1012: }
    1013:
    1014:
    1015: /**
    1016: * The fields of the certificate read from its DER, once: those the
    1017: * OpenSSL extension of PHP does not give.
    1018: *
    1019: * TBSCertificate: [0] version (absent: v1), serialNumber, signature,
    1020: * issuer, validity, subject, subjectPublicKeyInfo, [1] issuerUniqueID,
    1021: * [2] subjectUniqueID, [3] extensions. An extension the verification of
    1022: * a chain reads that cannot be decoded makes the certificate invalid, as
    1023: * for OpenSSL.
    1024: *
    1025: * @throws Exception If the DER cannot be read.
    1026: */
    1027: private function der()
    1028: {
    1029: if (null !== $this->der)
    1030: {
    1031: return $this->der;
    1032: }
    1033: try
    1034: {
    1035: $tbs = Der::childrenOf(Der::childrenOf(Der::readElement($this->getDer()), Der::SEQUENCE)[0], Der::SEQUENCE);
    1036: $version = 1;
    1037: if (isset($tbs[0]) && 0xa0 === $tbs[0]['tag'])
    1038: {
    1039: $version = Der::decodeSmallInteger(Der::readElement(array_shift($tbs)['content'])) + 1;
    1040: }
    1041: if (count($tbs) < 6 || Der::INTEGER !== $tbs[0]['tag'])
    1042: {
    1043: throw new Exception('DER: malformed certificate.');
    1044: }
    1045: $info = [
    1046: 'version' => $version,
    1047: 'serial' => $tbs[0]['content'],
    1048: 'signatureAlgorithm' => Der::decodeOid(Der::childrenOf($tbs[1], Der::SEQUENCE)[0]),
    1049: 'rawIssuer' => Der::reencode($tbs[2]),
    1050: 'issuer' => self::readName($tbs[2]),
    1051: 'rawSubject' => Der::reencode($tbs[4]),
    1052: 'subject' => self::readName($tbs[4]),
    1053: ];
    1054: $extensionsElement = null;
    1055: foreach (array_slice($tbs, 6) as $element)
    1056: {
    1057: if (0xa3 === $element['tag'])
    1058: {
    1059: $extensionsElement = $element;
    1060: }
    1061: }
    1062: list($info['extensions'], $info['invalidExtension']) = self::readExtensions($extensionsElement);
    1063: }
    1064: catch (Exception $e)
    1065: {
    1066: throw new Exception("Failed to extract X.509 information from certificate!");
    1067: }
    1068:
    1069: $info += ['keyUsage' => null, 'basicConstraints' => null, 'subjectKeyId' => null, 'authorityKeyId' => null];
    1070: $extensions = $info['extensions'];
    1071: try
    1072: {
    1073: if (isset($extensions[self::OID_KEY_USAGE]))
    1074: {
    1075: $info['keyUsage'] = Der::decodeBits(Der::readElement($extensions[self::OID_KEY_USAGE]['value']));
    1076: }
    1077: if (isset($extensions[self::OID_BASIC_CONSTRAINTS]))
    1078: {
    1079: $info['basicConstraints'] = self::readBasicConstraints($extensions[self::OID_BASIC_CONSTRAINTS]['value']);
    1080: // A path length without CA is invalid for OpenSSL.
    1081: if (null !== $info['basicConstraints']['pathLength'] && !$info['basicConstraints']['ca'])
    1082: {
    1083: $info['invalidExtension'] = true;
    1084: }
    1085: }
    1086: if (isset($extensions[self::OID_SUBJECT_KEY_ID]))
    1087: {
    1088: $element = Der::readElement($extensions[self::OID_SUBJECT_KEY_ID]['value']);
    1089: if (Der::OCTET_STRING !== $element['tag'])
    1090: {
    1091: throw new Exception('DER: malformed subject key identifier.');
    1092: }
    1093: $info['subjectKeyId'] = $element['content'];
    1094: }
    1095: if (isset($extensions[self::OID_AUTHORITY_KEY_ID]))
    1096: {
    1097: $info['authorityKeyId'] = self::readAuthorityKeyId($extensions[self::OID_AUTHORITY_KEY_ID]['value']);
    1098: }
    1099: }
    1100: catch (Exception $e)
    1101: {
    1102: $info['invalidExtension'] = true;
    1103: }
    1104:
    1105: return $this->der = $info;
    1106: }

    1109: /**
    1110: * Read the extensions: by OID, 'critical' and 'value'; and whether one
    1111: * is repeated.
    1112: */
    1113: private static function readExtensions($element)
    1114: {
    1115: $extensions = [];
    1116: $invalid = false;
    1117: if (null === $element)
    1118: {
    1119: return [$extensions, $invalid];
    1120: }
    1121: foreach (Der::childrenOf(Der::readElement($element['content']), Der::SEQUENCE) as $extension)
    1122: {
    1123: $fields = Der::childrenOf($extension, Der::SEQUENCE);
    1124: if (count($fields) < 2 || count($fields) > 3 || Der::OCTET_STRING !== $fields[count($fields) - 1]['tag'])
    1125: {
    1126: throw new Exception('DER: malformed extension.');
    1127: }
    1128: $oid = Der::decodeOid($fields[0]);
    1129: if (isset($extensions[$oid]))
    1130: {
    1131: $invalid = true;
    1132: }
    1133: $extensions[$oid] = [
    1134: 'critical' => 3 === count($fields) ? Der::decodeBoolean($fields[1]) : false,
    1135: 'value' => $fields[count($fields) - 1]['content'],
    1136: ];
    1137: }
    1138:
    1139: return [$extensions, $invalid];
    1140: }

    1143: /**
    1144: * Read the basic constraints: 'ca' and 'pathLength' (NULL if none).
    1145: */
    1146: private static function readBasicConstraints($value)
    1147: {
    1148: $ca = false;
    1149: $pathLength = null;
    1150: foreach (Der::childrenOf(Der::readElement($value), Der::SEQUENCE) as $field)
    1151: {
    1152: if (Der::BOOLEAN === $field['tag'] && null === $pathLength)
    1153: {
    1154: $ca = Der::decodeBoolean($field);
    1155: }
    1156: else
    1157: {
    1158: $pathLength = Der::decodeSmallInteger($field);
    1159: }
    1160: }
    1161:
    1162: return ['ca' => $ca, 'pathLength' => $pathLength];
    1163: }
    1164:
    1165:
    1166: /**
    1167: * Read the authority key identifier: [0] key identifier, [1] issuer (the
    1168: * first directory name of the general names, as OpenSSL takes it), [2]
    1169: * serial number of the issuer.
    1170: */
    1171: private static function readAuthorityKeyId($value)
    1172: {
    1173: $akid = ['keyId' => null, 'issuer' => null, 'serial' => null];
    1174: foreach (Der::childrenOf(Der::readElement($value), Der::SEQUENCE) as $field)
    1175: {
    1176: if (0x80 === $field['tag'])
    1177: {
    1178: $akid['keyId'] = $field['content'];
    1179: }
    1180: elseif (0xa1 === $field['tag'])
    1181: {
    1182: foreach (Der::readElements($field['content']) as $name)
    1183: {
    1184: if (0xa4 === $name['tag'])
    1185: {
    1186: $akid['issuer'] = self::readName(Der::readElement($name['content']));
    1187: break;
    1188: }
    1189: }
    1190: }
    1191: elseif (0x82 === $field['tag'])
    1192: {
    1193: $akid['serial'] = $field['content'];
    1194: }
    1195: else
    1196: {
    1197: throw new Exception('DER: malformed authority key identifier.');
    1198: }
    1199: }
    1200:
    1201: return $akid;
    1202: }
    1203:
    1204:
    1205: /**
    1206: * Read a name (a sequence of sets of fields), in the canonical form
    1207: * OpenSSL compares names by (`x509_name_canon()`): the values of the
    1208: * string types converted to UTF-8, without leading and trailing white
    1209: * space, each run of white space collapsed into one space, ASCII letters
    1210: * in lowercase; the values of other types kept as they are, with their
    1211: * type. The fields of an RDN are compared as a set. NULL if a value
    1212: * cannot be converted: the name then matches none.
    1213: */
    1214: private static function readName($element)
    1215: {
    1216: $rdns = [];
    1217: foreach (Der::childrenOf($element, Der::SEQUENCE) as $set => $rdn)
    1218: {
    1219: foreach (Der::childrenOf($rdn, Der::SET) as $attribute)
    1220: {
    1221: $parts = Der::childrenOf($attribute, Der::SEQUENCE);
    1222: if (2 !== count($parts))
    1223: {
    1224: throw new Exception('DER: malformed name.');
    1225: }
    1226: $oid = Der::decodeOid($parts[0]);
    1227: $value = $parts[1];
    1228: if (in_array($value['tag'], self::CANONICAL_STRING_TAGS, true))
    1229: {
    1230: $utf8 = self::toUtf8($value);
    1231: if (null === $utf8)
    1232: {
    1233: return null;
    1234: }
    1235: $text = 's:' . bin2hex(self::canonicalString($utf8));
    1236: }
    1237: else
    1238: {
    1239: $text = $value['tag'] . ':' . bin2hex($value['content']);
    1240: }
    1241: $rdns[$set][] = $oid . '=' . $text;
    1242: }
    1243: }
    1244: $canonical = [];
    1245: foreach ($rdns as $rdn)
    1246: {
    1247: sort($rdn, SORT_STRING);
    1248: $canonical[] = implode('+', $rdn);
    1249: }
    1250:
    1251: return implode('/', $canonical);
    1252: }
