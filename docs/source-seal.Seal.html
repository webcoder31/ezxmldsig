    81: /**
    82: * Verify a seal.
    83: *
    84: * The structure is read strictly: every element of the signature is
    85: * where the baseline puts it, once, and anything this library does not
    86: * process (a signature policy, a manifest of references, a transform)
    87: * makes the seal invalid rather than ignored. A seal is valid only if
    88: * every file of the container outside META-INF is covered, so that a
    89: * file added to a sealed container cannot pass for a sealed one.
    90: *
    91: * The signing certificate is trusted at the time of the signature
    92: * time-stamp, if any, and at the current date otherwise: SigningTime is
    93: * claimed by the signer, and cannot date the check of its certificate.
    94: * The certificate of the time-stamping authority is checked at the
    95: * current date. The revocation of the certificates is checked, by CRLs
    96: * or OCSP responses, unless 'revocation' is FALSE: the library sends no
    97: * request, the data come from the seal (level LT) and from the options.
    98: *
    99: * @param string $container The ASiC-E container.
    100: *
    101: * @param array $options
    102: * - 'caCertPath' (required): the trusted CA certificates, paths of
    103: * PEM files or PEM contents: the root of the signer, and the
    104: * intermediate CAs if the seal does not embed them;
    105: * - 'tsaCertPath': the trusted certificates of the time-stamping
    106: * authorities, required to verify a time-stamped seal;
    107: * - 'expectedSubject': the subject the signing certificate must
    108: * have, its DN as X509Cert::getDN() gives it, or fields that
    109: * must all match (see XMLDSigToken's option of the same name);
    110: * - 'revocation': whether the revocation is checked (default
    111: * TRUE);
    112: * - 'crls': CRLs, in DER or PEM, for the certificates of the
    113: * signer and of the time-stamping authority, used besides those
    114: * of a seal of level LT;
    115: * - 'ocspResponses': OCSP responses (DER), likewise;
    116: * - 'maxEntries', 'maxTotalSize': the limits of the reading of
    117: * the container (default 1000 entries, 512 MiB uncompressed).
    118: *
    119: * @return SealVerification The result. An invalid seal does not throw:
    120: * see its getError().
    121: *
    122: * @throws Exception On a misuse: the options are not valid, or a CA
    123: * certificate cannot be read.
    124: */
    125: public static function verifySeal($container, array $options)

    127: return self::inspectSeal($container, $options)['result'];

    132: * verifySeal(), with what the extension of a seal needs: the chains, the
    133: * certificates the signature and its time-stamp hold, and the sources
    134: * that told the status of each chain.

    138: public static function inspectSeal($container, array $options, $forLevelLT = false)
    139: {
    140: if (!is_string($container))
    141: {
    142: throw new Exception('Seal verification: the container must be bytes (a string).');
    143: }
    144: $caCerts = self::readCertificates(isset($options['caCertPath']) ? $options['caCertPath'] : null, 'caCertPath');
    145: $tsaCerts = isset($options['tsaCertPath']) ? self::readCertificates($options['tsaCertPath'], 'tsaCertPath') : null;
    146: $expectedSubject = isset($options['expectedSubject']) ? $options['expectedSubject'] : null;
    147: if (array_key_exists('expectedSubject', $options) && ((!is_string($expectedSubject) && !is_array($expectedSubject)) || '' === $expectedSubject || [] === $expectedSubject))
    148: {
    149: throw new Exception("Seal verification: invalid option 'expectedSubject'.");
    150: }
    151: $revocation = array_key_exists('revocation', $options) ? $options['revocation'] : true;
    152: if (!is_bool($revocation))
    153: {
    154: throw new Exception("Seal verification: invalid option 'revocation'.");
    155: }
    156: foreach (['crls' => 'CRLs', 'ocspResponses' => 'OCSP responses'] as $name => $what)
    157: {
    158: if (isset($options[$name]) && (!is_array($options[$name]) || count(array_filter($options[$name], 'is_string')) !== count($options[$name])))
    159: {
    160: throw new Exception("Seal verification: invalid option '" . $name . "': an array of " . $what . ' (strings).');
    161: }
    162: }
    163: $given = [];
    164: foreach (isset($options['crls']) ? $options['crls'] : [] as $crl)
    165: {
    166: try
    167: {
    168: $given[] = Crl::source(Crl::read($crl));
    169: }
    170: catch (Exception $e)
    171: {
    172: throw new Exception('Seal verification: a CRL cannot be used. ' . $e->getMessage());
    173: }
    174: }
    175: foreach (isset($options['ocspResponses']) ? $options['ocspResponses'] : [] as $response)
    176: {
    177: try
    178: {
    179: $given = array_merge($given, Ocsp::sources(Ocsp::read($response)));
    180: }
    181: catch (Exception $e)
    182: {
    183: throw new Exception('Seal verification: an OCSP response cannot be used. ' . $e->getMessage());
    184: }
    185: }
    186: $maxEntries = isset($options['maxEntries']) ? $options['maxEntries'] : Zip::MAX_ENTRIES;
    187: $maxTotalSize = isset($options['maxTotalSize']) ? $options['maxTotalSize'] : Zip::MAX_TOTAL_SIZE;
    188:
    189: $signer = null;
    190: $signingTime = null;
    191: $found = ['chain' => null, 'tsaChain' => null, 'keyInfo' => [], 'tokenCertificates' => [], 'signerSources' => [], 'tsaSources' => [], 'hasValidationData' => false];
    192: try
    193: {
    194: list($files, $signatureFile) = self::readContainer($container, $maxEntries, $maxTotalSize);
    195: $signature = self::readSignature($signatureFile, $files);
    196: $signer = $signature['signer'];
    197: $signingTime = $signature['signingTime'];
    198: $validationData = null === $signature['validationData'] ? null : self::readValidationData($signature['validationData']);
    199:
    200: // The signature time-stamp proves that the signature existed at
    201: // its time: the certificates of the signer are checked at that
    202: // time. The certificate of the authority is checked at the
    203: // current date: only a time-stamp over the time-stamp (level LTA)
    204: // would check it at an earlier one.
    205: $timestampTime = null;
    206: $tsaChain = null;
    207: $tokenCertificates = [];
    208: if (null !== $signature['timestamp'])
    209: {
    210: if (null === $tsaCerts)
    211: {
    212: self::invalid("The seal is time-stamped: the option 'tsaCertPath' is needed to trust its time-stamping authority.");
    213: }
    214: list($timestampTime, $tsaChain, $tokenCertificates) = self::checkTimestamp($signature['timestamp'], $tsaCerts, null === $validationData ? [] : $validationData['certificates']);
    215: }
    216: elseif (null !== $validationData)
    217: {
    218: self::invalid('The seal has validation data but no signature time-stamp: level LT is time-stamped.');
    219: }
    220:
    221: // Without time-stamp, the trust in the signer is checked at the
    222: // current date: SigningTime is claimed by the signer, and cannot
    223: // date the check of its certificate.
    224: $certificate = $signature['signer'];
    225: if (!$certificate->canSign())
    226: {
    227: self::invalid('The key usage of the signing certificate does not allow signing.');
    228: }
    229: $untrusted = array_merge($signature['chain'], null === $validationData ? [] : $validationData['certificates']);
    230: $chain = $certificate->getTrustedChain($caCerts, $untrusted, $timestampTime);
    231: if (null === $chain)
    232: {
    233: self::invalid('The signing certificate is not trusted: no valid chain up to a trusted CA certificate' . (null === $timestampTime ? '' : ' at the time of the time-stamp') . '.');
    234: }
    235: if (null !== $expectedSubject)
    236: {
    237: $matches = is_string($expectedSubject) ? $certificate->getDN() === $expectedSubject : X509Cert::matchesName($certificate->getSubject(), $expectedSubject);
    238: if (!$matches)
    239: {
    240: self::invalid('The subject of the signing certificate is not the expected one.');
    241: }
    242: }
    243: $found = ['chain' => $chain, 'tsaChain' => $tsaChain, 'keyInfo' => $signature['keyInfo'], 'tokenCertificates' => $tokenCertificates, 'hasValidationData' => null !== $validationData] + $found;
    244:
    245: if ($revocation)
    246: {
    247: // The signer: from the time of the time-stamp, if any. The
    248: // time-stamping authority: at the current date, unless the
    249: // seal embeds its validation data, which are no longer
    250: // current when it is verified.
    251: $now = time();
    252: $signerCheck = ['now' => $now, 'issuedAfter' => $timestampTime, 'time' => $timestampTime];
    253: $tsaCheck = ['now' => $now, 'issuedAfter' => null === $validationData && !$forLevelLT ? null : $timestampTime, 'time' => $timestampTime, 'authority' => true];
    254: $run = function (array $pool) use ($chain, $tsaChain, $signerCheck, $tsaCheck)
    255: {
    256: $ofSigner = Revocation::check($chain, $pool, $signerCheck);
    257: $ofAuthority = null === $tsaChain ? null : Revocation::check($tsaChain, $pool, $tsaCheck);
    258: $error = null !== $ofSigner['error'] ? $ofSigner['error'] : (null === $ofAuthority ? null : $ofAuthority['error']);
    259: return ['error' => $error, 'signerSources' => $ofSigner['used'], 'tsaSources' => null === $ofAuthority ? [] : $ofAuthority['used']];
    260: };
    261: if (null !== $validationData)
    262: {
    263: // Level LT: the embedded data must suffice by themselves.
    264: $own = $run($validationData['sources']);
    265: if (null !== $own['error'])
    266: {
    267: self::invalid('The validation data of the seal do not suffice. ' . $own['error']);
    268: }
    269: $found['signerSources'] = $own['signerSources'];
    270: $found['tsaSources'] = $own['tsaSources'];
    271: }
    272: if (null === $validationData || !empty($given))
    273: {
    274: $all = $run(array_merge(null === $validationData ? [] : $validationData['sources'], $given));
    275: if (null !== $all['error'])
    276: {
    277: self::invalid($all['error'] . (null === $validationData ? " (the option 'revocation' => false verifies the seal without revocation.)" : ''));
    278: }
    279: if (null === $validationData)
    280: {
    281: $found['signerSources'] = $all['signerSources'];
    282: $found['tsaSources'] = $all['tsaSources'];
    283: }
    284: }
    285: }
    286: $level = null === $timestampTime ? 'B' : (null === $validationData ? 'T' : 'LT');
    287:
    288: return ['result' => new SealVerification([
    289: 'error' => null,
    290: 'files' => $signature['files'],
    291: 'signingTime' => $signingTime,
    292: 'signer' => $signer,
    293: 'revocationChecked' => $revocation,
    294: 'timestampTime' => $timestampTime,
    295: 'level' => $level,
    296: ])] + $found;
    297: }
    298: catch (InvalidSeal $e)
    299: {
    300: return ['result' => new SealVerification(['error' => $e->getMessage(), 'signingTime' => $signingTime, 'signer' => $signer])] + $found;
    301: }
    302: }

    488: * signature file.

    490: private static function readContainer($archive, $maxEntries, $maxTotalSize)
    491: {
    492: try
    493: {
    494: $entries = Zip::read($archive, $maxEntries, $maxTotalSize);
    495: }
    496: catch (Exception $e)
    497: {
    498: self::invalid('The container is not a valid ZIP archive. ' . $e->getMessage());
    499: }
    500: $files = [];
    501: $signatureFiles = [];
    502: $manifest = null;
    503: foreach ($entries as $index => $entry)
    504: {
    505: if ('mimetype' === $entry['name'])
    506: {
    507: if (0 !== $index || 0 !== $entry['method'] || '' !== $entry['extra'])
    508: {
    509: self::invalid('The mimetype file must be the first entry, stored, without extra field.');
    510: }
    511: if (self::ASICE_MIMETYPE !== $entry['content'])
    512: {
    513: self::invalid('The container is not an ASiC-E container (mimetype ' . self::quote(self::latin1(substr($entry['content'], 0, 100))) . ').');
    514: }
    515: }
    516: elseif (0 === strpos($entry['name'], 'META-INF/'))
    517: {
    518: $base = strtolower(substr($entry['name'], strlen('META-INF/')));
    519: if (self::MANIFEST_PATH === $entry['name'])
    520: {
    521: $manifest = $entry;
    522: }
    523: elseif (false !== strpos($base, 'signatures') && '.xml' === substr($base, -4) && false === strpos($base, '/'))
    524: {
    525: $signatureFiles[] = $entry;
    526: }
    527: elseif (preg_match('/signature|timestamp|manifest|evidencerecord/', $base))
    528: {
    529: self::invalid('Unsupported file in META-INF: ' . self::quote($entry['name']) . ' (only XAdES signatures are supported).');
    530: }
    531: }
    532: else
    533: {
    534: $files[$entry['name']] = $entry['content'];
    535: }
    536: }
    537: if (null === $manifest)
    538: {
    539: self::invalid('The container has no META-INF/manifest.xml.');
    540: }
    541: self::checkManifest($manifest['content']);
